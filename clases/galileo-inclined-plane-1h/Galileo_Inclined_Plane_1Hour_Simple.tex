\begin{itemize}[leftmargin=5mm,itemsep=0pt,topsep=1pt]

\textbf{For every row}
\[
\bar t=\frac{t_1+t_2}{2}
\qquad\qquad
\bar t^{\,2}=(\bar t)^2
